% 基准：figure 编号 Figure N，caption 在下，居中
\documentclass[10pt]{article}
\usepackage{graphicx}
\usepackage{wrapfig}
\usepackage{subcaption}
\begin{document}
\begin{figure}[htbp]
  \centering
  \fbox{\rule{0pt}{2cm}\rule{2cm}{0pt}}
  \caption{Centered figure caption below}
  \label{fig:main}
\end{figure}

\begin{wrapfigure}{r}{0.4\textwidth}
  \fbox{\rule{0pt}{1.5cm}\rule{0.4\textwidth}{0pt}}
  \caption{Wrapped caption}
\end{wrapfigure}
Wrapped text paragraph that flows around the figure on the right side with columnsep spacing.

\begin{figure}[htbp]
  \centering
  \begin{subfigure}{0.48\textwidth}
    \centering
    \fbox{\rule{0pt}{1cm}\rule{0.45\textwidth}{0pt}}
    \caption{Sub caption A}
  \end{subfigure}
  \begin{subfigure}{0.48\textwidth}
    \centering
    \fbox{\rule{0pt}{1cm}\rule{0.45\textwidth}{0pt}}
    \caption{Sub caption B}
  \end{subfigure}
  \caption{Main caption}
\end{figure}
\end{document}
